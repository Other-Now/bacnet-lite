cmake_minimum_required(VERSION 3.16)
project(bacnet_lite CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

option(BL_SANITIZE "Build with ASan + UBSan" OFF)
option(BL_FUZZ "Build libFuzzer targets (clang only)" OFF)
option(BL_PLANT_BUG "Compile in the planted null-deref used by docs/CRASH_ANALYSIS.md" OFF)

if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS _WINSOCK_DEPRECATED_NO_WARNINGS)
else()
  add_compile_options(-Wall -Wextra -Wshadow -Wconversion -Wno-sign-conversion)
endif()

if(BL_SANITIZE AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=undefined)
  add_link_options(-fsanitize=address,undefined)
endif()

find_package(Threads REQUIRED)

add_library(bacnet src/codec.cpp src/protocol.cpp src/net.cpp src/objects.cpp src/stack.cpp
                   src/device.cpp src/client.cpp)
target_include_directories(bacnet PUBLIC src)
if(BL_PLANT_BUG)
  target_compile_definitions(bacnet PRIVATE BL_PLANT_BUG)
endif()
option(BL_KEEP_CONNRESET "Experiment: leave Winsock's UDP WSAECONNRESET reporting on" OFF)
if(BL_KEEP_CONNRESET)
  target_compile_definitions(bacnet PRIVATE BL_KEEP_CONNRESET)
endif()
target_link_libraries(bacnet PUBLIC Threads::Threads)
if(WIN32)
  target_link_libraries(bacnet PUBLIC ws2_32)
endif()

add_executable(bl-device apps/bl_device.cpp)
target_link_libraries(bl-device bacnet)
if(WIN32)
  target_link_libraries(bl-device dbghelp)
endif()
add_executable(bl-client apps/bl_client.cpp)
target_link_libraries(bl-client bacnet)
add_executable(bl-bench apps/bl_bench.cpp)
target_link_libraries(bl-bench bacnet)

add_executable(unit_tests tests/unit_tests.cpp)
target_link_libraries(unit_tests bacnet)
enable_testing()
add_test(NAME unit COMMAND unit_tests)

if(BL_FUZZ)
  foreach(t fuzz_device fuzz_client)
    add_executable(${t} fuzz/${t}.cpp)
    target_link_libraries(${t} bacnet)
    target_compile_options(${t} PRIVATE -fsanitize=fuzzer)
    target_link_options(${t} PRIVATE -fsanitize=fuzzer)
  endforeach()
  # the library itself needs coverage instrumentation too
  target_compile_options(bacnet PRIVATE -fsanitize=fuzzer-no-link)
endif()
